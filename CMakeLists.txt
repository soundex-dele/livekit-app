cmake_minimum_required(VERSION 3.24)
project(av_apps VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AV_BUILD_REMOTE_CONTROL_DESKTOP "Build the remote-control desktop product" ON)
option(AV_BUILD_CONFERENCE_DESKTOP "Build the conference desktop product" ON)
option(AV_BUILD_PLAYER_DESKTOP "Build the local-file player (requires FFmpeg and SDL development packages)" OFF)
include(CTest)
include(cmake/NativeSDK.cmake)
set(_av_rtc OFF)
if(AV_BUILD_REMOTE_CONTROL_DESKTOP OR AV_BUILD_CONFERENCE_DESKTOP)
  set(_av_rtc ON)
endif()
if(_av_rtc OR AV_BUILD_PLAYER_DESKTOP)
  av_add_native_sdk(${_av_rtc} ${AV_BUILD_PLAYER_DESKTOP})
endif()
if(AV_BUILD_REMOTE_CONTROL_DESKTOP)
  add_subdirectory(apps/remote-control/desktop)
endif()

if(AV_BUILD_CONFERENCE_DESKTOP)
  add_subdirectory(apps/conference/desktop)
endif()
if(AV_BUILD_PLAYER_DESKTOP)
  add_subdirectory(apps/player/desktop)
endif()
