cmake_minimum_required(VERSION 3.24)
project(av_apps VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
option(AV_BUILD_REMOTE_CONTROL_DESKTOP "Build the remote-control desktop product" ON)
option(AV_BUILD_SCREEN_CAST_DESKTOP "Build the screen-cast desktop product" ON)
option(AV_BUILD_CONFERENCE_DESKTOP "Build the conference desktop product" ON)
option(AV_BUILD_PLAYER_DESKTOP "Build the Qt6 local-file player (requires FFmpeg and Qt6 development packages)" OFF)
option(AV_BUILD_LIVE_STUDIO_DESKTOP "Build desktop live studio" OFF)
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  if(AV_BUILD_PLAYER_DESKTOP OR AV_BUILD_REMOTE_CONTROL_DESKTOP OR AV_BUILD_CONFERENCE_DESKTOP)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  else()
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
include(CTest)
include(cmake/NativeSDK.cmake)
set(_av_rtc OFF)
if(AV_BUILD_REMOTE_CONTROL_DESKTOP OR AV_BUILD_SCREEN_CAST_DESKTOP OR AV_BUILD_CONFERENCE_DESKTOP OR AV_BUILD_LIVE_STUDIO_DESKTOP)
  set(_av_rtc ON)
endif()
set(_av_player ${AV_BUILD_PLAYER_DESKTOP})
if(AV_BUILD_LIVE_STUDIO_DESKTOP)
  set(_av_player ON)
  set(LIVEKIT_BUILD_STUDIO ON CACHE BOOL "" FORCE)
endif()
if(_av_rtc OR _av_player)
  av_add_native_sdk(${_av_rtc} ${_av_player})
endif()
if(AV_BUILD_REMOTE_CONTROL_DESKTOP)
  add_subdirectory(apps/remote-control/desktop)
endif()

if(AV_BUILD_SCREEN_CAST_DESKTOP)
  add_subdirectory(apps/screen-cast/desktop)
endif()
if(AV_BUILD_CONFERENCE_DESKTOP)
  add_subdirectory(apps/conference/desktop)
endif()
if(AV_BUILD_PLAYER_DESKTOP)
  add_subdirectory(apps/player/desktop)
endif()
if(AV_BUILD_LIVE_STUDIO_DESKTOP)
  add_subdirectory(apps/live-studio/desktop)
endif()
