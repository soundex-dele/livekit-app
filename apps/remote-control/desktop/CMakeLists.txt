cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_remote_control_desktop LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  include(CTest)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "Remote-control desktop currently requires Windows")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
if(NOT TARGET LiveKit::SDK)
  get_filename_component(_sdk_default "${CMAKE_CURRENT_SOURCE_DIR}/../../../../livekit-sdk-cpp" ABSOLUTE)
  set(AV_SDK_SOURCE_DIR "${_sdk_default}" CACHE PATH "Native SDK checkout")
  set(LIVEKIT_BUILD_RTC ON CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_ENABLE_VIDEO_PLAYOUT ON CACHE BOOL "" FORCE)
  add_subdirectory("${AV_SDK_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/sdk")
endif()
include(FetchContent)
FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz
  URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(imgui)
add_library(remote_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(remote_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(remote_imgui PUBLIC SDL2::SDL2-static)
add_library(remote_host_session STATIC src/services/session.cpp src/infrastructure/http.cpp)
target_include_directories(remote_host_session PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(remote_host_session PUBLIC LiveKit::SDK PRIVATE winhttp)
target_compile_options(remote_host_session PRIVATE /utf-8)
add_executable(av_remote_host src/app/main.cpp src/features/remote_control/ui.cpp)
target_link_libraries(av_remote_host PRIVATE remote_host_session remote_imgui)
target_compile_options(av_remote_host PRIVATE /utf-8)
if(BUILD_TESTING)
add_executable(remote_host_render_test tests/render_test.cpp src/features/remote_control/ui.cpp)
target_link_libraries(remote_host_render_test PRIVATE remote_host_session remote_imgui)
add_test(NAME remote_host_render COMMAND remote_host_render_test)
set_tests_properties(remote_host_render PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 30)
add_executable(remote_host_fixture tests/fixture.cpp)
target_link_libraries(remote_host_fixture PRIVATE remote_host_session)

endif()
