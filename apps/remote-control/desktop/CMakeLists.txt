cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_remote_control_desktop VERSION 0.4.0 LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  if(NOT CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  endif()
  include(CTest)
endif()
if(NOT WIN32 AND NOT APPLE)
  message(FATAL_ERROR "Remote desktop supports Windows and macOS")
endif()
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Svg)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::SDK)
  set(LIVEKIT_ENABLE_VIDEO_PLAYOUT OFF CACHE BOOL "Qt owns remote video presentation" FORCE)
  av_add_native_sdk(ON OFF)
endif()
if(NOT TARGET av_http)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/http" "${CMAKE_BINARY_DIR}/av_http")
endif()
add_library(remote_host_session STATIC src/services/session.cpp)
target_include_directories(remote_host_session PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(remote_host_session PUBLIC LiveKit::SDK av_http)
add_library(remote_controller STATIC src/services/controller.cpp)
target_link_libraries(remote_controller PUBLIC remote_host_session Qt6::Gui)
add_library(remote_ui STATIC src/features/remote_control/ui.cpp src/features/remote_control/video_widget.cpp)
target_link_libraries(remote_ui PUBLIC remote_controller Qt6::Widgets Qt6::Svg)
if(APPLE)
  target_sources(remote_ui PRIVATE src/platform/platform_macos.mm)
  set_source_files_properties(src/platform/platform_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(remote_ui PRIVATE "-framework AppKit" "-framework Carbon" "-framework ApplicationServices")
else()
  target_sources(remote_ui PRIVATE src/platform/platform_windows.cpp)
endif()
add_executable(av_remote_host WIN32 MACOSX_BUNDLE src/app/main.cpp)
target_link_libraries(av_remote_host PRIVATE remote_ui)
if(APPLE)
  set_target_properties(av_remote_host PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.avplatform.remote-control MACOSX_BUNDLE_BUNDLE_NAME "AV Remote" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in")
endif()
if(MSVC)
  target_compile_options(remote_host_session PRIVATE /utf-8)
  target_compile_options(remote_controller PRIVATE /utf-8)
  target_compile_options(remote_ui PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(remote_lifecycle_test tests/lifecycle_test.cpp)
  target_link_libraries(remote_lifecycle_test PRIVATE remote_controller)
  add_test(NAME remote_lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lifecycle.py" "$<TARGET_FILE:remote_lifecycle_test>")
  set_tests_properties(remote_lifecycle PROPERTIES TIMEOUT 30)
  add_executable(remote_host_fixture tests/fixture.cpp)
  target_link_libraries(remote_host_fixture PRIVATE remote_host_session)
  add_executable(remote_controller_fixture tests/controller_fixture.cpp)
  target_link_libraries(remote_controller_fixture PRIVATE remote_controller)
  add_executable(remote_sdk_test "${AV_SDK_SOURCE_DIR}/tests/remote_control_tests.cpp")
  target_include_directories(remote_sdk_test PRIVATE "${AV_SDK_SOURCE_DIR}/src")
  target_link_libraries(remote_sdk_test PRIVATE LiveKit::SDK)
  add_test(NAME remote_sdk COMMAND remote_sdk_test "${CMAKE_CURRENT_SOURCE_DIR}/../../../../docs/integration/remote-control/messages.json")
  add_executable(remote_protocol_test tests/protocol_test.cpp)
  target_include_directories(remote_protocol_test PRIVATE "${AV_SDK_SOURCE_DIR}/src")
  target_link_libraries(remote_protocol_test PRIVATE remote_ui Qt6::Test)
  add_test(NAME remote_protocol COMMAND remote_protocol_test)
  add_executable(remote_host_render_test tests/render_test.cpp)
  target_link_libraries(remote_host_render_test PRIVATE remote_ui Qt6::Test)
  add_test(NAME remote_host_render COMMAND remote_host_render_test)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/render-hidpi")
  add_test(NAME remote_host_render_hidpi COMMAND remote_host_render_test)
  set_tests_properties(remote_host_render_hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2" WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/render-hidpi" TIMEOUT 30)
  set_tests_properties(remote_host_render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
install(TARGETS av_remote_host BUNDLE DESTINATION . COMPONENT RemoteControl RUNTIME DESTINATION bin COMPONENT RemoteControl)
qt_generate_deploy_app_script(TARGET av_remote_host OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${qt_deploy} COMPONENT RemoteControl)

if(APPLE)
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    # Homebrew exports a relative symlink; install its actual binary so the
    # headless verification plugin remains valid after moving the application.
    install(CODE "
      file(REAL_PATH \"$<TARGET_FILE:Qt6::QOffscreenIntegrationPlugin>\" _offscreen)
      set(_destination \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/av_remote_host.app/Contents/PlugIns/platforms\")
      file(REMOVE \"\${_destination}/libqoffscreen.dylib\")
      file(INSTALL \"\${_offscreen}\" DESTINATION \"\${_destination}\" USE_SOURCE_PERMISSIONS)
    " COMPONENT RemoteControl)
  endif()
  configure_file(macos/InstallBundle.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/InstallRemoteBundle.cmake" @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallRemoteBundle.cmake" COMPONENT RemoteControl)
endif()
