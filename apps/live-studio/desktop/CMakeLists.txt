cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_live_studio_desktop VERSION 0.3.0 LANGUAGES C CXX)
  include(CTest)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::Studio)
  set(LIVEKIT_BUILD_STUDIO ON CACHE BOOL "" FORCE)
  av_add_native_sdk(ON ON)
endif()
if(NOT TARGET av_imgui)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/imgui" "${CMAKE_BINARY_DIR}/av_imgui")
endif()
if(NOT TARGET av_http)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/http" "${CMAKE_BINARY_DIR}/av_http")
endif()
add_library(studio_platform STATIC)
target_include_directories(studio_platform PUBLIC src)
target_link_libraries(studio_platform PUBLIC LiveKit::SDK)
if(APPLE)
  target_sources(studio_platform PRIVATE src/platform_macos.mm)
  target_compile_options(studio_platform PRIVATE -fobjc-arc)
  target_link_libraries(studio_platform PRIVATE "-framework AppKit" "-framework Security" "-framework LocalAuthentication" "-framework UniformTypeIdentifiers")
else()
  target_sources(studio_platform PRIVATE src/platform_windows.cpp)
  target_link_libraries(studio_platform PRIVATE crypt32 comdlg32 shell32)
  target_compile_definitions(studio_platform PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(av_live_studio src/main.cpp src/session.cpp)
target_link_libraries(av_live_studio PRIVATE LiveKit::Studio av_imgui av_http ixwebsocket studio_platform)
target_compile_definitions(av_live_studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
livekit_player_runtime(av_live_studio)
if(APPLE)
  set_target_properties(av_live_studio PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.avplatform.live-studio"
    MACOSX_BUNDLE_BUNDLE_NAME "AV Live Studio"
    MACOSX_BUNDLE_BUNDLE_VERSION "0.3.0"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.0")
  install(TARGETS av_live_studio BUNDLE DESTINATION . COMPONENT LiveStudio)
  configure_file(macos/InstallBundle.cmake.in InstallStudioBundle.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallStudioBundle.cmake" COMPONENT LiveStudio)
endif()
if(BUILD_TESTING)
  add_executable(studio_media_graph_tests "${AV_SDK_SOURCE_DIR}/tests/media_graph_tests.cpp")
  target_include_directories(studio_media_graph_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src")
  target_link_libraries(studio_media_graph_tests PRIVATE LiveKit::SDK livekit_media_nodes_webrtc livekit_rtc_graph LiveKit::MediaGraph graphflow::core WebRTC::webrtc)
  add_test(NAME studio_media_graph COMMAND studio_media_graph_tests)
  set_tests_properties(studio_media_graph PROPERTIES TIMEOUT 60)
  add_executable(studio_publisher_fixture tests/publisher_fixture.cpp src/session.cpp)
  target_link_libraries(studio_publisher_fixture PRIVATE LiveKit::Studio av_http ixwebsocket studio_platform)
  livekit_player_runtime(studio_publisher_fixture)
  add_executable(studio_engine_tests tests/engine_test.cpp)
  target_include_directories(studio_engine_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src/engines/studio")
  target_link_libraries(studio_engine_tests PRIVATE LiveKit::Studio)
  livekit_player_runtime(studio_engine_tests)
  if(APPLE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(STUDIO_FFPROBE ffprobe HINTS "${LIVEKIT_FFMPEG_ROOT}/bin" REQUIRED)
    find_program(STUDIO_FFMPEG ffmpeg HINTS "${LIVEKIT_FFMPEG_ROOT}/bin" REQUIRED)
    add_test(NAME studio_engine COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_engine_test.py"
      "$<TARGET_FILE:studio_engine_tests>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" "${STUDIO_FFPROBE}" "${STUDIO_FFMPEG}")
  else()
    add_test(NAME studio_engine COMMAND studio_engine_tests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  endif()
  set_tests_properties(studio_engine PROPERTIES TIMEOUT 90)
  if(WIN32)
    add_executable(studio_graph_tests tests/graph_test.cpp)
    target_include_directories(studio_graph_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src")
    target_link_libraries(studio_graph_tests PRIVATE LiveKit::Studio graphflow::engine ole32)
    livekit_player_runtime(studio_graph_tests)
    add_test(NAME studio_graph COMMAND studio_graph_tests "${CMAKE_CURRENT_BINARY_DIR}/graph-fixtures")
    set_tests_properties(studio_graph PROPERTIES TIMEOUT 45)
  endif()
  add_test(NAME studio_ui COMMAND av_live_studio --hidden --frames 60 --settings "${CMAKE_CURRENT_BINARY_DIR}/ui-fixture" --screenshot "${CMAKE_CURRENT_BINARY_DIR}/studio.bmp")
  set_tests_properties(studio_ui PROPERTIES TIMEOUT 45 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()
