cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_live_studio_desktop VERSION 0.3.0 LANGUAGES C CXX)
  include(CTest)
endif()
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::Studio)
  set(LIVEKIT_BUILD_STUDIO ON CACHE BOOL "" FORCE)
  av_add_native_sdk(ON ON)
endif()
if(NOT TARGET av_http)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/http" "${CMAKE_BINARY_DIR}/av_http")
endif()
add_library(studio_platform STATIC)
target_include_directories(studio_platform PUBLIC src)
target_link_libraries(studio_platform PUBLIC LiveKit::SDK)
if(APPLE)
  target_sources(studio_platform PRIVATE src/platform_macos.mm)
  target_compile_options(studio_platform PRIVATE -fobjc-arc)
  target_link_libraries(studio_platform PRIVATE "-framework AppKit" "-framework Security" "-framework LocalAuthentication" "-framework UniformTypeIdentifiers")
else()
  target_sources(studio_platform PRIVATE src/platform_windows.cpp)
  target_link_libraries(studio_platform PRIVATE crypt32 comdlg32 shell32)
  target_compile_definitions(studio_platform PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(studio_widgets STATIC src/window.cpp src/preview.cpp src/session.cpp)
target_link_libraries(studio_widgets PUBLIC LiveKit::Studio Qt6::Widgets Qt6::Concurrent av_http ixwebsocket studio_platform)
target_compile_definitions(studio_widgets PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN QT_NO_KEYWORDS)
qt_add_executable(av_live_studio WIN32 src/main.cpp)
target_link_libraries(av_live_studio PRIVATE studio_widgets)
livekit_player_runtime(av_live_studio)
if(WIN32)
  find_program(STUDIO_WINDEPLOYQT windeployqt HINTS "${QT6_INSTALL_PREFIX}/bin" REQUIRED)
  add_custom_command(TARGET av_live_studio POST_BUILD
    COMMAND "${STUDIO_WINDEPLOYQT}" --no-translations --no-opengl-sw --no-system-d3d-compiler "$<TARGET_FILE:av_live_studio>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${QT6_INSTALL_PREFIX}/plugins/platforms/qoffscreen$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:av_live_studio>/platforms/"
    VERBATIM)
endif()
if(APPLE)
  set_target_properties(av_live_studio PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.avplatform.live-studio"
    MACOSX_BUNDLE_BUNDLE_NAME "AV Live Studio"
    MACOSX_BUNDLE_BUNDLE_VERSION "0.3.0"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.0")
  install(TARGETS av_live_studio BUNDLE DESTINATION . COMPONENT LiveStudio)
endif()
if(WIN32)
  install(TARGETS av_live_studio RUNTIME DESTINATION . COMPONENT LiveStudio)
  install(DIRECTORY "${LIVEKIT_FFMPEG_ROOT}/bin/" DESTINATION . COMPONENT LiveStudio FILES_MATCHING PATTERN "*.dll")
  # Keep the historical flat Windows package layout, including non-Qt DLLs
  # placed beside the executable by vcpkg's application-local deployment.
  install(CODE "file(GLOB _studio_dlls \"$<TARGET_FILE_DIR:av_live_studio>/*.dll\")
    file(INSTALL \${_studio_dlls} DESTINATION \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}\")
    set(QT_DEPLOY_BIN_DIR \".\")" COMPONENT LiveStudio)
  install(FILES "${QT6_INSTALL_PREFIX}/plugins/platforms/qoffscreen$<$<CONFIG:Debug>:d>.dll"
    DESTINATION plugins/platforms COMPONENT LiveStudio)
endif()
qt_generate_deploy_app_script(TARGET av_live_studio OUTPUT_SCRIPT studio_deploy)
install(SCRIPT ${studio_deploy} COMPONENT LiveStudio)
if(APPLE)
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    get_target_property(STUDIO_OFFSCREEN_PLUGIN Qt6::QOffscreenIntegrationPlugin LOCATION)
    install(FILES "${STUDIO_OFFSCREEN_PLUGIN}" DESTINATION av_live_studio.app/Contents/PlugIns/platforms COMPONENT LiveStudio)
  endif()
  configure_file(macos/InstallBundle.cmake.in InstallStudioBundle.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallStudioBundle.cmake" COMPONENT LiveStudio)
endif()
if(BUILD_TESTING)
  add_executable(studio_media_graph_tests "${AV_SDK_SOURCE_DIR}/tests/media_graph_tests.cpp")
  target_include_directories(studio_media_graph_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src")
  target_link_libraries(studio_media_graph_tests PRIVATE LiveKit::SDK livekit_media_nodes_webrtc livekit_rtc_graph LiveKit::MediaGraph graphflow::core WebRTC::webrtc)
  add_test(NAME studio_media_graph COMMAND studio_media_graph_tests)
  set_tests_properties(studio_media_graph PROPERTIES TIMEOUT 60)
  add_executable(studio_publisher_fixture tests/publisher_fixture.cpp src/session.cpp)
  target_link_libraries(studio_publisher_fixture PRIVATE LiveKit::Studio av_http ixwebsocket studio_platform)
  livekit_player_runtime(studio_publisher_fixture)
  add_executable(studio_engine_tests tests/engine_test.cpp)
  target_include_directories(studio_engine_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src/engines/studio")
  target_link_libraries(studio_engine_tests PRIVATE LiveKit::Studio)
  livekit_player_runtime(studio_engine_tests)
  if(APPLE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(STUDIO_FFPROBE ffprobe HINTS "${LIVEKIT_FFMPEG_ROOT}/bin" REQUIRED)
    find_program(STUDIO_FFMPEG ffmpeg HINTS "${LIVEKIT_FFMPEG_ROOT}/bin" REQUIRED)
    add_test(NAME studio_engine COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_engine_test.py"
      "$<TARGET_FILE:studio_engine_tests>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" "${STUDIO_FFPROBE}" "${STUDIO_FFMPEG}")
  else()
    add_test(NAME studio_engine COMMAND studio_engine_tests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  endif()
  set_tests_properties(studio_engine PROPERTIES TIMEOUT 90)
  if(WIN32)
    add_executable(studio_graph_tests tests/graph_test.cpp)
    target_include_directories(studio_graph_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src")
    target_link_libraries(studio_graph_tests PRIVATE LiveKit::Studio graphflow::engine ole32)
    livekit_player_runtime(studio_graph_tests)
    add_test(NAME studio_graph COMMAND studio_graph_tests "${CMAKE_CURRENT_BINARY_DIR}/graph-fixtures")
    set_tests_properties(studio_graph PROPERTIES TIMEOUT 45)
  endif()
  add_test(NAME studio_ui COMMAND av_live_studio --hidden --frames 60 --settings "${CMAKE_CURRENT_BINARY_DIR}/ui-fixture" --screenshot "${CMAKE_CURRENT_BINARY_DIR}/studio.bmp")
  set_tests_properties(studio_ui PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_executable(studio_widgets_tests tests/widgets_test.cpp)
  set_target_properties(studio_widgets_tests PROPERTIES AUTOMOC ON)
  target_link_libraries(studio_widgets_tests PRIVATE studio_widgets Qt6::Test)
  livekit_player_runtime(studio_widgets_tests)
  if(WIN32)
    add_custom_command(TARGET studio_widgets_tests POST_BUILD
      COMMAND "${STUDIO_WINDEPLOYQT}" --no-translations --no-opengl-sw --no-system-d3d-compiler "$<TARGET_FILE:studio_widgets_tests>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${QT6_INSTALL_PREFIX}/plugins/platforms/qoffscreen$<$<CONFIG:Debug>:d>.dll" "$<TARGET_FILE_DIR:studio_widgets_tests>/platforms/"
      VERBATIM)
  endif()
  add_test(NAME studio_widgets COMMAND studio_widgets_tests -o -,txt)
  set_tests_properties(studio_widgets PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME studio_widgets_hidpi COMMAND studio_widgets_tests portraitAndLayout -o -,txt)
  set_tests_properties(studio_widgets_hidpi PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2")
  add_test(NAME studio_ui_portrait COMMAND av_live_studio --hidden --portrait --frames 45 --settings "${CMAKE_CURRENT_BINARY_DIR}/ui-portrait-fixture" --screenshot "${CMAKE_CURRENT_BINARY_DIR}/studio-portrait.png")
  add_test(NAME studio_ui_hidpi COMMAND av_live_studio --hidden --test-density 2 --frames 45 --settings "${CMAKE_CURRENT_BINARY_DIR}/ui-hidpi-fixture" --screenshot "${CMAKE_CURRENT_BINARY_DIR}/studio-hidpi.png")
  set_tests_properties(studio_ui_portrait studio_ui_hidpi PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
