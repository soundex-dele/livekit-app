cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_freecut_desktop VERSION 0.1.0 LANGUAGES C CXX)
  include(CTest)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Multimedia)
if(NOT TARGET av_ui_qt)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/qt" "${CMAKE_BINARY_DIR}/av_ui_qt")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::Editor)
  if(TARGET LiveKit::MediaCore)
    message(FATAL_ERROR "Enable LIVEKIT_BUILD_EDITOR before adding the SDK")
  endif()
  set(LIVEKIT_BUILD_EDITOR ON CACHE BOOL "" FORCE)
  set(LIVEKIT_EDITOR_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "" FORCE)
  av_add_native_sdk(OFF OFF)
endif()
set(CMAKE_AUTOMOC ON)
add_library(freecut_app STATIC window.cpp timeline.cpp ../../player/desktop/audio_output.cpp)
target_include_directories(freecut_app PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(freecut_app PUBLIC LiveKit::Editor Qt6::Widgets Qt6::Multimedia av_ui_qt)
qt_add_executable(freecut WIN32 MACOSX_BUNDLE main.cpp)
target_link_libraries(freecut PRIVATE freecut_app)
livekit_player_runtime(freecut)
include(GNUInstallDirs)
if(MSVC)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
  include(InstallRequiredSystemLibraries)
endif()
install(TARGETS freecut BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
qt_generate_deploy_app_script(TARGET freecut OUTPUT_SCRIPT freecut_deploy_script)
install(SCRIPT ${freecut_deploy_script})
if(WIN32)
  install(DIRECTORY "${LIVEKIT_FFMPEG_ROOT}/bin/" DESTINATION ${CMAKE_INSTALL_BINDIR} FILES_MATCHING PATTERN "*.dll")
  install(FILES "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_PLUGINS}/platforms/qoffscreen$<$<CONFIG:Debug>:d>.dll"
    DESTINATION "${QT6_INSTALL_PLUGINS}/platforms")
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_executable(freecut_ui_tests tests/ui_tests.cpp)
  target_link_libraries(freecut_ui_tests PRIVATE freecut_app Qt6::Test)
  target_compile_definitions(freecut_ui_tests PRIVATE
    FREECUT_FIXTURES="${CMAKE_BINARY_DIR}/sdk/src/engines/editor/fixtures"
    FREECUT_SCREENSHOTS="${CMAKE_CURRENT_BINARY_DIR}/screenshots")
  livekit_player_runtime(freecut_ui_tests)
  add_test(NAME freecut_widgets COMMAND freecut_ui_tests)
  set_tests_properties(freecut_widgets PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_QPA_PLATFORM=offscreen" FIXTURES_REQUIRED editor_media)
  add_test(NAME freecut_widgets_hidpi COMMAND freecut_ui_tests empty)
  set_tests_properties(freecut_widgets_hidpi PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2")
  get_target_property(_qt_core_location Qt6::Core IMPORTED_LOCATION_RELEASE)
  get_filename_component(_qt_bin "${_qt_core_location}" DIRECTORY)
  set_tests_properties(freecut_widgets freecut_widgets_hidpi PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qt_bin}")
  if(WIN32)
    set_property(TEST freecut_widgets freecut_widgets_hidpi APPEND PROPERTY
      ENVIRONMENT_MODIFICATION "QT_QPA_FONTDIR=set:$ENV{WINDIR}/Fonts")
  endif()
endif()
