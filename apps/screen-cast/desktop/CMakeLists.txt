cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_screen_cast_desktop LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  include(CTest)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "Screen-cast desktop currently requires Windows")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::SDK)
  av_add_native_sdk(ON OFF)
endif()
if(NOT TARGET av_imgui)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/imgui" "${CMAKE_BINARY_DIR}/av_imgui")
endif()
if(NOT TARGET av_http)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/http" "${CMAKE_BINARY_DIR}/av_http")
endif()
add_library(screen_cast_session STATIC src/services/session.cpp)
target_include_directories(screen_cast_session PUBLIC src)
target_link_libraries(screen_cast_session PUBLIC LiveKit::SDK av_http)
target_compile_options(screen_cast_session PRIVATE /utf-8)
add_executable(av_screen_cast src/app/main.cpp src/features/ui.cpp)
target_link_libraries(av_screen_cast PRIVATE screen_cast_session av_imgui)
target_compile_options(av_screen_cast PRIVATE /utf-8)
if(BUILD_TESTING)
  add_executable(screen_cast_render_test tests/render_test.cpp src/features/ui.cpp)
  target_link_libraries(screen_cast_render_test PRIVATE screen_cast_session av_imgui)
  target_compile_options(screen_cast_render_test PRIVATE /utf-8)
  add_test(NAME screen_cast_render COMMAND screen_cast_render_test)
  set_tests_properties(screen_cast_render PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 30)
  add_executable(screen_cast_fixture tests/fixture.cpp)
  target_link_libraries(screen_cast_fixture PRIVATE screen_cast_session)
  add_executable(screen_cast_session_test tests/session_test.cpp)
  target_link_libraries(screen_cast_session_test PRIVATE screen_cast_session)
  add_test(NAME screen_cast_session COMMAND screen_cast_session_test)
  set_tests_properties(screen_cast_session PROPERTIES TIMEOUT 30)
endif()
