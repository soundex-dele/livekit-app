cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_screen_cast_desktop VERSION 0.3.0 LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  endif()
  include(CTest)
endif()
if(NOT WIN32 AND NOT APPLE)
  message(FATAL_ERROR "Screen-cast desktop supports Windows and macOS")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
option(SCREEN_CAST_UI_ONLY "Build isolated Qt UI tests without native media dependencies" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
if(MSVC)
  if((QT_FEATURE_static_runtime AND CMAKE_MSVC_RUNTIME_LIBRARY MATCHES "DLL$") OR
     (NOT QT_FEATURE_static_runtime AND NOT CMAKE_MSVC_RUNTIME_LIBRARY MATCHES "DLL$"))
    message(FATAL_ERROR "Qt6 and the SDK must use the same MSVC runtime. Configure CMAKE_MSVC_RUNTIME_LIBRARY and WEBRTC_PREBUILT_DIR to match Qt6.")
  endif()
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
set(_sdk "${AV_SDK_SOURCE_DIR}")
add_library(screen_cast_ui STATIC src/features/ui.cpp src/features/ui.hpp src/services/controller.cpp)
set_target_properties(screen_cast_ui PROPERTIES AUTOMOC ON)
target_include_directories(screen_cast_ui PUBLIC src "${_sdk}/include")
target_link_libraries(screen_cast_ui PUBLIC Qt6::Widgets)
if(MSVC)
  target_compile_options(screen_cast_ui PRIVATE /utf-8)
endif()
if(NOT SCREEN_CAST_UI_ONLY)
  if(NOT TARGET LiveKit::RTCInputs)
    # This Widgets client consumes frame callbacks and never opens SDL windows.
    # Configure its standalone SDK without pulling in an unused SDL renderer.
    if(APPLE)
      enable_language(OBJC OBJCXX)
      set(CMAKE_OBJCXX_STANDARD 20)
      set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    endif()
        set(LIVEKIT_BUILD_RTC_NATIVE ON CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_RTC_GRAPH_NODES ON CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_RTC_MEDIA ON CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_PLAYER OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_STUDIO OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_SERVER_SDK ON CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_CLI OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LIVEKIT_ENABLE_VIDEO_PLAYOUT OFF CACHE BOOL "" FORCE)
    add_subdirectory("${AV_SDK_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/sdk")
  endif()
  if(NOT TARGET av_http)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../packages/native/http" "${CMAKE_BINARY_DIR}/av_http")
  endif()
  add_library(screen_cast_session STATIC src/services/session.cpp)
  target_include_directories(screen_cast_session PUBLIC src)
  target_link_libraries(screen_cast_session PUBLIC LiveKit::RTCMediaEngine av_http)
  if(APPLE)
    target_sources(screen_cast_session PRIVATE src/platform/permission_macos.mm)
    target_link_libraries(screen_cast_session PRIVATE "-framework CoreGraphics")
  else()
    target_sources(screen_cast_session PRIVATE src/platform/permission_windows.cpp)
    target_compile_options(screen_cast_session PRIVATE /utf-8)
  endif()
  qt_add_executable(av_screen_cast src/app/main.cpp src/services/native_backend.cpp)
  target_link_libraries(av_screen_cast PRIVATE screen_cast_ui screen_cast_session)
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    qt_import_plugins(av_screen_cast INCLUDE Qt6::QOffscreenIntegrationPlugin)
  endif()
  if(MSVC)
    target_compile_options(av_screen_cast PRIVATE /utf-8)
    set_target_properties(av_screen_cast PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  if(APPLE)
    set_target_properties(av_screen_cast PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in"
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.avplatform.screen-cast")
  endif()
  install(TARGETS av_screen_cast
    BUNDLE DESTINATION . COMPONENT ScreenCast
    RUNTIME DESTINATION . COMPONENT ScreenCast)
  if(APPLE)
    get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION_RELEASE)
    if(NOT _qmake)
      get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION)
    endif()
    execute_process(COMMAND "${_qmake}" -query QT_INSTALL_LIBS
      OUTPUT_VARIABLE SCREEN_CAST_QT_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    # Homebrew installs Qt modules in separate prefixes; give macdeployqt the
    # unified library location so optional image-format plugins resolve QtSvg.
    qt_generate_deploy_app_script(TARGET av_screen_cast OUTPUT_SCRIPT _deploy
      DEPLOY_TOOL_OPTIONS "-libpath=${SCREEN_CAST_QT_LIBS}")
  else()
    qt_generate_deploy_app_script(TARGET av_screen_cast OUTPUT_SCRIPT _deploy NO_UNSUPPORTED_PLATFORM_ERROR)
  endif()
  install(SCRIPT "${_deploy}" COMPONENT ScreenCast)
  if(APPLE)
    get_target_property(_offscreen Qt6::QOffscreenIntegrationPlugin IMPORTED_LOCATION_RELEASE)
    if(NOT _offscreen)
      get_target_property(_offscreen Qt6::QOffscreenIntegrationPlugin IMPORTED_LOCATION)
    endif()
    get_filename_component(SCREEN_CAST_QT_OFFSCREEN "${_offscreen}" REALPATH)
    configure_file(macos/InstallBundle.cmake.in InstallScreenCast.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallScreenCast.cmake" COMPONENT ScreenCast)
  endif()
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_executable(screen_cast_render_test tests/render_test.cpp)
  set_target_properties(screen_cast_render_test PROPERTIES AUTOMOC ON)
  target_link_libraries(screen_cast_render_test PRIVATE screen_cast_ui Qt6::Test)
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    qt_import_plugins(screen_cast_render_test INCLUDE Qt6::QOffscreenIntegrationPlugin)
  endif()
  if(MSVC)
    target_compile_options(screen_cast_render_test PRIVATE /utf-8)
  endif()
  add_test(NAME screen_cast_render COMMAND screen_cast_render_test)
  set_tests_properties(screen_cast_render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
  add_test(NAME screen_cast_render_hidpi COMMAND screen_cast_render_test)
  set_tests_properties(screen_cast_render_hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2;SC_SCREENSHOT_SUFFIX=-2x" TIMEOUT 45)
  if(NOT SCREEN_CAST_UI_ONLY)
    add_test(NAME screen_cast_runtime COMMAND av_screen_cast --check-runtime)
    set_tests_properties(screen_cast_runtime PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    add_executable(screen_cast_capture_test "${AV_SDK_SOURCE_DIR}/tests/screen_capture_tests.cpp")
    target_include_directories(screen_cast_capture_test PRIVATE "${AV_SDK_SOURCE_DIR}/src")
    target_link_libraries(screen_cast_capture_test PRIVATE LiveKit::MediaInputs)
    add_test(NAME screen_cast_capture COMMAND screen_cast_capture_test)
    add_executable(screen_cast_fixture tests/fixture.cpp)
    target_link_libraries(screen_cast_fixture PRIVATE screen_cast_session)
    add_executable(screen_cast_session_test tests/session_test.cpp)
    target_link_libraries(screen_cast_session_test PRIVATE screen_cast_session)
    add_test(NAME screen_cast_session COMMAND screen_cast_session_test)
    set_tests_properties(screen_cast_session PROPERTIES TIMEOUT 30)
  endif()
endif()
