cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(av_conference_desktop VERSION 0.3.0 LANGUAGES C CXX)
  include(CTest)
endif()
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/NativeSDK.cmake")
if(NOT TARGET LiveKit::SDK)
  set(LIVEKIT_ENABLE_VIDEO_PLAYOUT OFF CACHE BOOL "" FORCE)
  av_add_native_sdk(ON OFF)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_library(livekit_demo_session STATIC session.cpp chat.cpp)
target_include_directories(livekit_demo_session PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(livekit_demo_session PUBLIC LiveKit::SDK)
add_library(av_meeting_widgets STATIC qt/api.cpp qt/api.hpp qt/video_widget.cpp qt/video_widget.hpp qt/window.cpp qt/window.hpp)
target_link_libraries(av_meeting_widgets PUBLIC livekit_demo_session Qt6::Widgets Qt6::Network)
add_executable(livekit_conference_demo main.cpp)
target_link_libraries(livekit_conference_demo PRIVATE av_meeting_widgets)
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set_target_properties(livekit_conference_demo PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.livekit.conference-demo"
    MACOSX_BUNDLE_BUNDLE_NAME "AV 会议"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
endif()
install(TARGETS livekit_conference_demo
  BUNDLE DESTINATION . COMPONENT Conference
  RUNTIME DESTINATION bin COMPONENT Conference)
set(_av_meeting_deploy_options)
if(APPLE)
  # Homebrew installs optional Qt modules outside qtbase. Its image plugin may
  # depend on QtSvg even though this Widgets application does not link it.
  find_path(AV_MEETING_QT_EXTRA_LIBRARY_DIR QtSvg.framework
    PATHS /usr/local/lib /opt/homebrew/lib NO_DEFAULT_PATH)
  if(AV_MEETING_QT_EXTRA_LIBRARY_DIR)
    list(APPEND _av_meeting_deploy_options "-libpath=${AV_MEETING_QT_EXTRA_LIBRARY_DIR}")
  endif()
endif()
qt_generate_deploy_app_script(TARGET livekit_conference_demo OUTPUT_SCRIPT qt_deploy
  DEPLOY_TOOL_OPTIONS ${_av_meeting_deploy_options})
install(SCRIPT ${qt_deploy} COMPONENT Conference)
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  # Copy non-system dylibs (OpenSSL, libvpx, dav1d) and rewrite their install
  # names so the installed app does not depend on the developer's Homebrew.
  if(TARGET Qt6::QOffscreenIntegrationPlugin)
    get_target_property(AV_MEETING_OFFSCREEN_PLUGIN Qt6::QOffscreenIntegrationPlugin LOCATION)
  endif()
  configure_file(macos/InstallBundle.cmake.in InstallBundle.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallBundle.cmake" COMPONENT Conference)
elseif(WIN32)
  configure_file(windows/InstallMedia.cmake.in InstallMedia.in.cmake @ONLY)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/InstallMedia-$<CONFIG>.cmake"
    INPUT "${CMAKE_CURRENT_BINARY_DIR}/InstallMedia.in.cmake")
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallMedia-$<CONFIG>.cmake" COMPONENT Conference)
endif()
if(BUILD_TESTING)
  add_executable(livekit_conference_chat_tests tests/chat_tests.cpp)
  target_link_libraries(livekit_conference_chat_tests PRIVATE livekit_demo_session)
  add_test(NAME conference_chat COMMAND livekit_conference_chat_tests)
  set_tests_properties(conference_chat PROPERTIES TIMEOUT 30)
  add_executable(livekit_conference_chat_integration tests/chat_integration.cpp)
  target_link_libraries(livekit_conference_chat_integration PRIVATE livekit_demo_session)
  add_executable(livekit_demo_integration tests/demo_integration.cpp)
  target_link_libraries(livekit_demo_integration PRIVATE livekit_demo_session)
  add_executable(livekit_conference_business_integration tests/business_integration.cpp)
  target_link_libraries(livekit_conference_business_integration PRIVATE livekit_demo_session Qt6::Network)
  add_executable(av_meeting_audio_device_tests "${AV_SDK_SOURCE_DIR}/tests/media_graph_tests.cpp")
  target_include_directories(av_meeting_audio_device_tests PRIVATE "${AV_SDK_SOURCE_DIR}/src")
  target_link_libraries(av_meeting_audio_device_tests PRIVATE LiveKit::SDK livekit_media_nodes_webrtc livekit_rtc_graph LiveKit::MediaGraph graphflow::core WebRTC::webrtc)
  add_test(NAME conference_audio_devices COMMAND av_meeting_audio_device_tests)
  set_tests_properties(conference_audio_devices PROPERTIES TIMEOUT 45)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  add_executable(livekit_demo_tests tests/qt_tests.cpp)
  target_link_libraries(livekit_demo_tests PRIVATE av_meeting_widgets Qt6::Test)
  add_test(NAME conference_widgets COMMAND livekit_demo_tests)
  set_tests_properties(conference_widgets PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
